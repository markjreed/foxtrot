MODULE PUNISHMENT (IDENT='V1.0', MAIN=PUNISH,
                    ADDRESSING_MODE(EXTERNAL=GENERAL)) =
BEGIN

GLOBAL ROUTINE PUNISH =
BEGIN
    LIBRARY 'SYS$LIBRARY:STARLET';
    EXTERNAL ROUTINE LIB$PUT_OUTPUT;

    INCR count FROM 1 TO 500 DO
    BEGIN
        LIB$PUT_OUTPUT(%ASCID 'I will not throw paper airplanes in class.');
    END;

    RETURN SS$_NORMAL;
END;

END
